const LBL = 20;
const colw = w => 21 + 5 * (w - 1);
const colEnd = w => 25 + 5 * (w - 1);

function put(parts) {
  const arr = [];
  for (const [c, t] of parts) {
    const chars = [...t];
    for (let k = 0; k < chars.length; k++) {
      const idx = c - 1 + k;
      while (arr.length <= idx) arr.push(' ');
      if (arr[idx] !== ' ') throw new Error('collision at column ' + (idx + 1) + ': "' + arr[idx] + '" vs "' + chars[k] + '"');
      arr[idx] = chars[k];
    }
  }
  return arr.join('');
}
function label(t) {
  if (t.length > LBL) throw new Error('label too long (' + t.length + '): ' + t);
  return t + ' '.repeat(LBL - t.length);
}
function bar(labelText, segs) {
  const parts = [[1, label(labelText)]];
  for (const [a, b, ch] of segs) parts.push([colw(a), ch.repeat(colEnd(b) - colw(a) + 1)]);
  return put(parts);
}
